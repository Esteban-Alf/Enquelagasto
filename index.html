-- ---------- MOVIMIENTOS (ingresos / gastos) ---------- create table if not exists public.movimientos ( id uuid primary key default gen_random_uuid(), user_id uuid not null references auth.users(id) on delete cascade, tipo text not null check (tipo in ('ingreso','gasto')), importe numeric not null, categoria_id uuid references public.categorias(id) on delete set null, comentario text, ambito text not null default 'individual' check (ambito in ('individual','compartido')), es_fijo boolean default false, frecuencia text default 'unico', -- unico | semanal | mensual | anual fecha date not null default current_date, fecha_vencimiento date, -- opcional, para recordatorios foto text, -- base64 (o más adelante URL de Storage) created_at timestamptz default now() ); -- ---------- COMPRAS EN CUOTAS ---------- create table if not exists public.compras_tarjeta ( id uuid primary key default gen_random_uuid(), user_id uuid not null references auth.users(id) on delete cascade, tarjeta_id uuid not null references public.tarjetas(id) on delete cascade, descripcion text not null, monto_total numeric not null, cuotas int not null default 1, primera_cuota date not null default current_date, -- mes/fecha de la 1ª cuota categoria_id uuid references public.categorias(id) on delete set null, ambito text not null default 'individual' check (ambito in ('individual','compartido')), created_at timestamptz default now() );
